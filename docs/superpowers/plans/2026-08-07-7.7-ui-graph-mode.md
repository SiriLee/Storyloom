```python
@staticmethod
def _stub_process_factory():
    """§7.6 stub — MATCH returns real stub asset ID; GENERATE fills placeholder.

    Relies on pre-populated stub entries (done in mount_graph_pipeline).
    §7.8 deletes this entire method and replaces with real LLM matching.
    """
    import time
    from storyloom.tasks import TaskType

    def _factory(asset_type, local_name, roster):
        _TARGET = {
            AssetType.CHAR_PORTRAIT: "stub_default_portrait",
            AssetType.BACKGROUND: "stub_default_background",
        }.get(asset_type, "stub_default_background")

        def _process(task):
            time.sleep(0.01)
            if task.task_type is TaskType.MATCH:
                task.result = local_name  # result is local_name, not target
            else:  # GENERATE: fill the placeholder created by TaskGenerator
                roster.set_target(asset_type, local_name, _TARGET)
        return _process

    return _factory
```

```javascript
    /** Flatten engine choices to {key, label, enabled, reason}[].  Matches Display.flattenChoices(). */
    function _flattenChoices(choices) {
        const result = [];
        let idx = 0;
        for (const c of choices) {
            const labels = c.labels || [];
            const enabled = c.enabled || labels.map(() => true);
            const branches = c.branches || [];
            const conditions = c.conditions || {};
            for (let i = 0; i < labels.length; i++) {
                idx++;
                const isEnabled = i < enabled.length ? enabled[i] : true;
                let reason = "";
                if (!isEnabled) {
                    const branch = branches[i] || "";
                    const cond = conditions[branch] || "";
                    if (cond) {
                        const match = cond.match(/^\s*([\p{L}\p{N}_]+)\s*(==|!=|>=|<=|>|<)\s*(.+?)\s*$/u);
                        if (match) {
                            const varName = match[1];
                            const current = GameState.stateVars[varName];
                            if (current !== undefined) {
                                reason = _("Requires {cond}, current: {val}")
                                    .replace("{cond}", cond).replace("{val}", String(current));
                            }
                        }
                        if (!reason) reason = _("Requires {cond}").replace("{cond}", cond);
                    }
                }
                result.push({ key: String(idx), label: labels[i], enabled: isEnabled, reason: reason });
            }
        }
        return result;
    }

    /* ── Keyboard handler ───────────────────────────────────────── */

    function _onKeyDown(e) {
        if (e.key === " " || e.key === "Enter") {
            e.preventDefault();
            if (_immersive) { setImmersive(false); return; }
            if (_mode === "manual") _advance();
        }
        if (e.key === "Escape") {
            hideSettings();
            hideBacklog();
        }
        if (e.key === "a" && !e.ctrlKey && !e.metaKey) setAutoMode(_mode !== "auto");
        if (e.key === "i" && !e.ctrlKey && !e.metaKey) setImmersive(!_immersive);
        if (e.key === "b" && !e.ctrlKey && !e.metaKey) {
            const bl = $("#vnBacklog");
            if (bl && bl.style.display === "flex") hideBacklog(); else showBacklog();
        }
    }

        if (el) el.textContent = title;

        setTitle,
```
